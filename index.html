let currentColor = '#000000'; const grid = document.getElementById('grid'); const palette = document.getElementById('palette'); const cells = []; // 1. Build the 16x16 grid for (let i = 0; i < 256; i++) { const cell = document.createElement('div'); cell.className = 'cell'; cell.style.backgroundColor = '#FFFFFF'; const paint = (e) => { e.preventDefault(); cell.style.backgroundColor = currentColor; }; cell.addEventListener('mousedown', paint); cell.addEventListener('mouseenter', (e) => { if (e.buttons === 1) paint(e); }); cell.addEventListener('touchstart', paint); cell.addEventListener('touchmove', (e) => { e.preventDefault(); const touch = e.touches[0]; const element = document.elementFromPoint(touch.clientX, touch.clientY); if (element && element.classList.contains('cell')) { element.style.backgroundColor = currentColor; } }); grid.appendChild(cell); cells.push(cell); } // 2. Build the Color Palette const paletteColors = ['#000000', '#FFFFFF', '#8B4513', '#A0522D', '#32CD32', '#1E90FF', '#FFD700', '#FF69B4']; paletteColors.forEach(color => { const btn = document.createElement('div'); btn.className = 'color'; btn.style.backgroundColor = color; if (color === currentColor) btn.classList.add('selected'); btn.onclick = () => { currentColor = color; document.querySelectorAll('.color').forEach(c => c.classList.remove('selected')); btn.classList.add('selected'); }; palette.appendChild(btn); }); // 3.
